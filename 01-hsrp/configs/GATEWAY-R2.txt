! GATEWAY-R2 (Cisco 1841) - HSRP Standby (priority default 100)
enable
configure terminal
hostname GATEWAY-R2

interface FastEthernet0/1
 description Uplink ke ISP
 ip address dhcp
 ip nat outside
 no shutdown

interface FastEthernet0/0
 description LAN 192.168.1.0/24
 ip address 192.168.1.2 255.255.255.0
 ip nat inside
 standby 1 ip 192.168.1.3
 no shutdown

access-list 77 permit 192.168.1.0 0.0.0.255
ip nat inside source list 77 interface FastEthernet0/1 overload
ip route 0.0.0.0 0.0.0.0 200.1.1.1
end
write memory

! show ip nat translations
